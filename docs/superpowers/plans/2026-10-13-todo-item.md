```java
package com.antdigital.todo;

import java.time.Instant;
import java.util.UUID;

/**
 * 待办事项领域实体
 * <p>
 * 不可变对象，构造时自动生成唯一 ID 与创建时间。
 * 名称不能为空或仅含空白；描述可为 null，存储时转为空字符串。
 * </p>
 *
 * @author aiwork
 * @date 2026/10/13
 */
public final class TodoItem {

    private final String id;
    private final String name;
    private final String description;
    private final Instant createdAt;

    /**
     * 构造待办事项
     *
     * @param name        事项名称，不能为空或仅含空白
     * @param description 事项描述，可为 null（存储为空字符串）
     * @throws IllegalArgumentException 当 name 为 null、空字符串或仅含空白时
     */
    public TodoItem(String name, String description) {
        if (name == null || name.isBlank()) {
            throw new IllegalArgumentException("事项名称不能为空");
        }
        this.id = UUID.randomUUID().toString();
        this.name = name.trim();
        this.description = description == null ? "" : description.trim();
        this.createdAt = Instant.now();
    }

    public String getId() {
        return id;
    }

    public String getName() {
        return name;
    }

    public String getDescription() {
        return description;
    }

    public Instant getCreatedAt() {
        return createdAt;
    }
}
```

```java
package com.antdigital.todo;

import java.util.List;

/**
 * 待办事项仓库端口
 * <p>
 * 定义持久化操作的抽象接口，允许替换不同存储实现。
 * </p>
 *
 * @author aiwork
 * @date 2026/10/13
 */
public interface TodoRepository {

    /**
     * 保存待办事项
     *
     * @param item 待保存的待办事项，不能为 null
     * @return 保存后的待办事项（原样返回）
     */
    TodoItem save(TodoItem item);

    /**
     * 查询所有已保存的待办事项
     *
     * @return 待办事项列表的防御性副本
     */
    List<TodoItem> findAll();
}
```

- [ ] **Step 4: 实现内存仓库**

```java
package com.antdigital.todo;

import java.util.ArrayList;
import java.util.List;

/**
 * 内存待办事项仓库实现
 * <p>
 * 基于 ArrayList 的简单内存存储，适用于测试和最小闭环验证。
 * </p>
 *
 * @author aiwork
 * @date 2026/10/13
 */
public class InMemoryTodoRepository implements TodoRepository {

    private final List<TodoItem> store = new ArrayList<>();

    @Override
    public TodoItem save(TodoItem item) {
        store.add(item);
        return item;
    }

    @Override
    public List<TodoItem> findAll() {
        return new ArrayList<>(store);
    }
}
```

```java
package com.antdigital.todo;

/**
 * 待办事项应用服务
 * <p>
 * 封装"创建待办事项"用例，协调领域对象的校验与持久化。
 * </p>
 *
 * @author aiwork
 * @date 2026/10/13
 */
public class TodoService {

    private final TodoRepository repository;

    /**
     * 构造应用服务
     *
     * @param repository 待办事项仓库，不能为 null
     */
    public TodoService(TodoRepository repository) {
        this.repository = repository;
    }

    /**
     * 创建新的待办事项并持久化
     *
     * @param name        事项名称，不能为空或仅含空白
     * @param description 事项描述，可为 null
     * @return 创建并保存后的待办事项
     * @throws IllegalArgumentException 当 name 为 null、空字符串或仅含空白时
     */
    public TodoItem createTodo(String name, String description) {
        TodoItem item = new TodoItem(name, description);
        return repository.save(item);
    }
}
```
